## Reproducing H1 and M1 without paid services

Place the following temporary file at `internal/pipeline/reviewer_pin_test.go` in either candidate context. Run `go test ./internal/pipeline -run TestReviewer -v` with local test sockets allowed, then remove the file. The fixture helper uses only local servers. Inspect the printed key, cache hit, query and collected identity rather than treating a PASS banner as evidence.

```go
package pipeline

import (
    "context"
    "fmt"
    "testing"
    "github.com/nrynss/before-yes/internal/appstore"
    "github.com/nrynss/before-yes/internal/checklist"
)

func TestReviewerLegacyCache(t *testing.T) {
    e := newE2E(t, demos[0], review355())
    ctx := context.Background()
    key := EvidenceKey("Roblox", "us", e.p.Reader.Model.Settings().Model)
    e.p.Cache.Put(ctx, key, Evidence{Title:"Roblox", Kind:KindIOS,
        App:&AppInfo{TrackID:431946152, Name:"Roblox"}})
    ev, hit, err := e.p.Collect(ctx, "Roblox", "review")
    if err != nil { t.Fatal(err) }
    card := Card{Title:ev.Title, Kind:ev.Kind, Age:9,
        Verdict:checklist.No, App:ev.App}
    fmt.Printf("VERSION=%s KEY=%s HIT=%v GENRE=%q QUERY=%q\n",
        GatherVersion(), key, hit, ev.App.Genre, alternativesSearches(card)[0].Query)
}

type reviewIdentityStore struct { Store }
func (s reviewIdentityStore) Storefront() string { return "us" }
func (s reviewIdentityStore) Find(ctx context.Context, title string, limit int) ([]appstore.App,error) {
    if title == "Calm Builder" {
        return []appstore.App{{ID:3333, Name:"Calm Builder Companion"},
            {ID:2222, Name:"Calm Builder"}}, nil
    }
    return s.Store.Find(ctx,title,limit)
}
func TestReviewerConfirmedIdentity(t *testing.T) {
    e, _, _, card := altE2E(t)
    e.p.Gatherer.Store = reviewIdentityStore{e.p.Gatherer.Store}
    pending, err := e.p.FindAlternatives(context.Background(),card,"identity")
    if err != nil { t.Fatal(err) }
    for _, a := range pending {
        if a.Title != "Calm Builder" { continue }
        ev, _, err := e.p.Collect(context.Background(),a.Title,"identity-check")
        if err != nil { t.Fatal(err) }
        fmt.Printf("pending=%q confirmed_id=2222 collected_id=%d collected_name=%q\n",
            a.Title,ev.App.TrackID,ev.App.Name)
    }
}
```
